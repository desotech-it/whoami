WhoAmI has been deployed!

{{- if .Values.ingress.enabled }}

Access the application at:
{{- range $host := .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}
{{- end }}

{{- else if contains "NodePort" .Values.service.type }}

Get the application URL:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "whoami.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT

{{- else if contains "LoadBalancer" .Values.service.type }}

Get the application URL:
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "whoami.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}

{{- else if contains "ClusterIP" .Values.service.type }}

Access the application with port-forward:
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "whoami.fullname" . }} 8080:{{ .Values.service.port }}
  echo "Visit http://localhost:8080"

{{- end }}

Quick test:
  curl http://localhost:8080/
